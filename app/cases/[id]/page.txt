import { notFound } from "next/navigation";

import { prisma } from "@/lib/prisma";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { EditCaseForm } from "../components/edit-case-form";
import { updateCaseAction } from "./actions";

type Props = {
  params: Promise<{ id: string }>;
};

export default async function CaseDetailsPage({ params }: Props) {
  const { id } = await params;

  const [caseItem, clinics, serviceTypes, dentists] = await Promise.all([
    prisma.case.findUnique({
      where: { id },
      include: {
        clinic: true,
        dentist: true,
        serviceType: true,
        cadDesigner: true,
        createdByUser: true,
        caseComponentUsages: {
          include: {
            component: true,
          },
          orderBy: { createdAt: "desc" },
        },
        millings: {
          include: {
            blockType: true,
            millingDrill: true,
          },
          orderBy: { milledAt: "desc" },
        },
      },
    }),
    prisma.clinic.findMany({ orderBy: { name: "asc" } }),
    prisma.serviceType.findMany({
      where: { isActive: true },
      orderBy: { name: "asc" },
    }),
    prisma.dentist.findMany({
      orderBy: { name: "asc" },
      include: { clinic: true },
    }),
  ]);

  if (!caseItem) {
    notFound();
  }

  return (
    <main className="mx-auto max-w-6xl space-y-6 p-6">
      <div className="space-y-1">
        <h1 className="text-2xl font-semibold">
          {caseItem.code} — {caseItem.patientName}
        </h1>
        <p className="text-sm text-muted-foreground">
          Manage the case data from one place.
        </p>
      </div>

      <EditCaseForm
        action={updateCaseAction}
        caseItem={caseItem}
        clinics={clinics}
        serviceTypes={serviceTypes}
        dentists={dentists}
      />

      <section className="grid gap-6 lg:grid-cols-2">
        <Card className="rounded-2xl">
          <CardHeader>
            <CardTitle className="text-base">Components used</CardTitle>
          </CardHeader>
          <CardContent className="space-y-3 text-sm">
            {caseItem.caseComponentUsages.length === 0 ? (
              <p className="text-muted-foreground">No components yet.</p>
            ) : (
              caseItem.caseComponentUsages.map((usage) => (
                <div key={usage.id} className="rounded-xl border p-3">
                  <p className="font-medium">{usage.component.name}</p>
                  <p>Quantity: {usage.quantity}</p>
                  <p>Charge client: {usage.chargeClient ? "Yes" : "No"}</p>
                  <p>Notes: {usage.notes ?? "-"}</p>
                </div>
              ))
            )}
          </CardContent>
        </Card>

        <Card className="rounded-2xl">
          <CardHeader>
            <CardTitle className="text-base">Milling history</CardTitle>
          </CardHeader>
          <CardContent className="space-y-3 text-sm">
            {caseItem.millings.length === 0 ? (
              <p className="text-muted-foreground">No milling records yet.</p>
            ) : (
              caseItem.millings.map((milling) => (
                <div key={milling.id} className="rounded-xl border p-3">
                  <p className="font-medium">
                    {milling.blockType.name} — {milling.status}
                  </p>
                  <p>Teeth milled: {milling.teethMilledQty}</p>
                  <p>Drill: {milling.millingDrill?.name ?? "-"}</p>
                  <p>Failure reason: {milling.failureReason ?? "-"}</p>
                  <p>Notes: {milling.notes ?? "-"}</p>
                </div>
              ))
            )}
          </CardContent>
        </Card>
      </section>
    </main>
  );
}
